jsmn Quick Reference
Basic Setup
c#include "jsmn.h"
#include <string.h>

jsmn_parser parser;
jsmntok_t tokens[128];  // Pre-allocate token array

jsmn_init(&parser);
int r = jsmn_parse(&parser, json_str, strlen(json_str), tokens, 128);

if (r < 0) {
    // Error: JSMN_ERROR_INVAL, JSMN_ERROR_NOMEM, JSMN_ERROR_PART
}
// r = number of tokens parsed
Token Types
ctypedef enum {
    JSMN_UNDEFINED = 0,
    JSMN_OBJECT = 1,    // { }
    JSMN_ARRAY = 2,     // [ ]
    JSMN_STRING = 3,    // "string"
    JSMN_PRIMITIVE = 4  // number, true, false, null
} jsmntype_t;
Token Structure
ctypedef struct {
    jsmntype_t type;
    int start;   // Start position in JSON string
    int end;     // End position in JSON string
    int size;    // Number of children (object keys/array items)
    int parent;  // Parent token index (-1 if root)
} jsmntok_t;
Helper Function (Essential!)
c// Compare token with string
int jsoneq(const char *json, jsmntok_t *tok, const char *s) {
    if (tok->type == JSMN_STRING && 
        (int)strlen(s) == tok->end - tok->start &&
        strncmp(json + tok->start, s, tok->end - tok->start) == 0) {
        return 0;
    }
    return -1;
}
Common Patterns
Simple Object
c// JSON: {"name":"value", "count":42}

for (int i = 1; i < r; i++) {
    if (jsoneq(json, &tokens[i], "name") == 0) {
        // tokens[i+1] is the value
        printf("Name: %.*s\n", 
               tokens[i+1].end - tokens[i+1].start,
               json + tokens[i+1].start);
        i++; // Skip value token
    }
    else if (jsoneq(json, &tokens[i], "count") == 0) {
        int count = atoi(json + tokens[i+1].start);
        i++;
    }
}
Array
c// JSON: {"items":[1,2,3,4]}

if (jsoneq(json, &tokens[i], "items") == 0) {
    int array_size = tokens[i+1].size;  // Number of elements
    int j;
    for (j = 0; j < array_size; j++) {
        jsmntok_t *elem = &tokens[i+j+2];
        printf("Item %d: %.*s\n", j,
               elem->end - elem->start,
               json + elem->start);
    }
    i += array_size + 1;  // Skip array and its elements
}
Nested Object
c// JSON: {"config":{"port":8080,"host":"localhost"}}

if (jsoneq(json, &tokens[i], "config") == 0) {
    i++; // Move to the object token
    int obj_size = tokens[i].size;  // Number of key-value pairs
    i++; // Move to first key
    
    for (int j = 0; j < obj_size; j++) {
        if (jsoneq(json, &tokens[i], "port") == 0) {
            int port = atoi(json + tokens[i+1].start);
            i += 2;
        }
        else if (jsoneq(json, &tokens[i], "host") == 0) {
            // Extract host string
            i += 2;
        }
    }
}

Error Codes
c#define JSMN_ERROR_NOMEM -1  // Not enough tokens
#define JSMN_ERROR_INVAL -2  // Invalid JSON
#define JSMN_ERROR_PART  -3  // Incomplete JSON (streaming)
Pro Tips

Always check return value - negative means error, positive is token count
Token array sizing - better too big than too small (will return JSMN_ERROR_NOMEM)
No string termination - jsmn doesn't null-terminate, use start/end offsets
Zero-copy - jsmn just indexes into your original string
Reusable parser - call jsmn_init() to reset for next parse

Perfect for your 32KB Cortex-M devices - the parser itself is tiny and you control all memory allocation!Retry
