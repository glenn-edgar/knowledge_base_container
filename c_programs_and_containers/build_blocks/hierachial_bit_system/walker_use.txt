#include <stdio.h>

#include "cfl_hbit2_api.h"

/* ============================================ */
/* Example 1: Simple Iterator                   */
/* ============================================ */

void example_iterator(cfl_hbit2_tree_t* tree, int32_t root, int16_t bs_id) {
    printf("=== Iterator Pattern ===\n");
    
    cfl_hbit2_error_walker_t walker;
    cfl_hbit2_error_walker_init(&walker, tree, root, bs_id, true);  /* use_mask=true */
    
    int32_t node;
    int bit;
    
    while (cfl_hbit2_error_walker_next(&walker, &node, &bit) == CFL_HBIT2_WALK_OK) {
        const char* name = cfl_hbit2_info_name(tree, node);
        printf("  Error: %s (node %d) bit %d\n", name ? name : "?", node, bit);
    }
    
    printf("\n");
}

/* ============================================ */
/* Example 2: Callback with Count               */
/* ============================================ */

static bool print_error_cb(cfl_hbit2_tree_t* tree, int32_t node, int bit, void* user) {
    (void)user;
    const char* name = cfl_hbit2_info_name(tree, node);
    printf("  [CB] Error: %s bit %d\n", name ? name : "?", bit);
    return true;  /* continue */
}

void example_callback(cfl_hbit2_tree_t* tree, int32_t root, int16_t bs_id) {
    printf("=== Callback Pattern ===\n");
    
    cfl_hbit2_error_walker_t walker;
    cfl_hbit2_error_walker_init(&walker, tree, root, bs_id, true);
    
    int count = cfl_hbit2_error_walker_foreach(&walker, print_error_cb, NULL);
    printf("  Total errors: %d\n\n", count);
}

/* ============================================ */
/* Example 4: Collect Errors Into Array         */
/* ============================================ */

typedef struct {
    int32_t node;
    int bit;
} error_entry_t;

typedef struct {
    error_entry_t* entries;
    int count;
    int max;
} error_list_t;

static bool collect_error_cb(cfl_hbit2_tree_t* tree, int32_t node, int bit, void* user) {
    (void)tree;
    error_list_t* list = (error_list_t*)user;
    
    if (list->count < list->max) {
        list->entries[list->count].node = node;
        list->entries[list->count].bit = bit;
        list->count++;
        return true;  /* continue */
    }
    return false;  /* stop, array full */
}

void example_collect(cfl_hbit2_tree_t* tree, int32_t root, int16_t bs_id) {
    printf("=== Collect Into Array ===\n");
    
    error_entry_t entries[32];
    error_list_t list = { entries, 0, 32 };
    
    cfl_hbit2_error_walker_t walker;
    cfl_hbit2_error_walker_init(&walker, tree, root, bs_id, true);
    cfl_hbit2_error_walker_foreach(&walker, collect_error_cb, &list);
    
    printf("  Collected %d errors:\n", list.count);
    for (int i = 0; i < list.count; i++) {
        const char* name = cfl_hbit2_info_name(tree, list.entries[i].node);
        printf("    [%d] %s bit %d\n", i, name ? name : "?", list.entries[i].bit);
    }
    printf("\n");
}

/* ============================================ */
/* Example 5: Stop After First N Errors         */
/* ============================================ */

typedef struct {
    int max_errors;
    int found;
} limit_ctx_t;

static bool limit_error_cb(cfl_hbit2_tree_t* tree, int32_t node, int bit, void* user) {
    limit_ctx_t* ctx = (limit_ctx_t*)user;
    const char* name = cfl_hbit2_info_name(tree, node);
    
    printf("  Error %d: %s bit %d\n", ctx->found + 1, name ? name : "?", bit);
    ctx->found++;
    
    return (ctx->found < ctx->max_errors);  /* stop after max */
}

void example_limit(cfl_hbit2_tree_t* tree, int32_t root, int16_t bs_id, int max) {
    printf("=== First %d Errors ===\n", max);
    
    limit_ctx_t ctx = { max, 0 };
    
    cfl_hbit2_error_walker_t walker;
    cfl_hbit2_error_walker_init(&walker, tree, root, bs_id, true);
    cfl_hbit2_error_walker_foreach(&walker, limit_error_cb, &ctx);
    
    printf("  Showed %d errors\n\n", ctx.found);
}

/* ============================================ */
/* Example 7: Compare Masked vs Unmasked        */
/* ============================================ */

void example_masked_vs_unmasked(cfl_hbit2_tree_t* tree, int32_t root, int16_t bs_id) {
    printf("=== Masked vs Unmasked ===\n");
    
    cfl_hbit2_error_walker_t walker;
    
    /* Count with mask */
    cfl_hbit2_error_walker_init(&walker, tree, root, bs_id, true);
    int masked_count = cfl_hbit2_error_walker_foreach(&walker, NULL, NULL);
    
    /* Count without mask */
    cfl_hbit2_error_walker_init(&walker, tree, root, bs_id, false);
    int unmasked_count = cfl_hbit2_error_walker_foreach(&walker, NULL, NULL);
    
    printf("  Active errors (unmasked): %d\n", masked_count);
    printf("  Total errors (all):       %d\n", unmasked_count);
    printf("  Suppressed by mask:       %d\n\n", unmasked_count - masked_count);
}

/* ============================================ */
/* Example 8: Walk Both OR and AND Bitspaces    */
/* ============================================ */

void example_both_bitspaces(cfl_hbit2_tree_t* tree) {
    printf("=== Both Bitspaces ===\n");
    
    int32_t or_root = cfl_hbit2_node(tree, "Overall_Valve_Status");
    int32_t and_root = cfl_hbit2_node(tree, "Overall_Valve_State");
    
    cfl_hbit2_error_walker_t walker;
    
    /* OR bitspace (alarms) */
    if (or_root >= 0) {
        cfl_hbit2_error_walker_init(&walker, tree, or_root, 0, true);
        int count = cfl_hbit2_error_walker_foreach(&walker, NULL, NULL);
        printf("  OR bitspace (alarms):     %d errors\n", count);
    }
    
    /* AND bitspace (ready) */
    if (and_root >= 0) {
        cfl_hbit2_error_walker_init(&walker, tree, and_root, 1, true);
        int count = cfl_hbit2_error_walker_foreach(&walker, NULL, NULL);
        printf("  AND bitspace (not ready): %d errors\n", count);
    }
    
    printf("\n");
}
